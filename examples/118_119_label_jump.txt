# label / jump – Sprungmarken setzen und anspringen

# Einfacher Sprung nach vorne (Abschnitt überspringen)
if switch(1) == ON:
    jump("skip_intro")
endif
show_text("Das hier wird übersprungen wenn Schalter 1 AN ist.")
play_se("Cursor", 80, 100)
label("skip_intro")
show_text("Weiter geht's hier.")

# Sprung nach hinten (schlechtes Beispiel, besser loop verwenden)
variable(1) = 0
label("repeat_section")
variable(1) += 1
show_text("Durchlauf!")
if variable(1) < 3:
    jump("repeat_section")
endif
show_text("3 Durchläufe abgeschlossen.")

# Mehrere Labels in einem Script
label("start")
show_choices("Weiter", "Ende", cancel=-1):
    when("Weiter"):
        show_text("Nächster Abschnitt.")
        jump("middle")
    when("Ende"):
        jump("finish")
end_choices

label("middle")
show_text("Mitte des Scripts.")
jump("finish")

label("finish")
show_text("Ende des Scripts.")
