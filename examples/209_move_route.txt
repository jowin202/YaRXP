# move_route – Bewegungsroute für Charakter festlegen
# Ziel: player | this | event(ID)
# Optionen: repeat (Loop), skip (überspringen wenn blockiert)
# wait_for_move wartet bis die Route abgeschlossen ist

# Einfache Route für dieses Event
move_route(this):
    move_right
    move_right
    move_up
    move_up
    move_left
    turn_down
end_move_route
wait_for_move

# Route mit Wiederholung und Skip
move_route(this, repeat, skip):
    move_random
    wait(30)
end_move_route

# Route für den Spieler
move_route(player):
    move_down
    move_down
    move_right
    turn_up
end_move_route
wait_for_move
show_text("Du bist angekommen.")

# Route für Event 3 (NPC geht weg)
move_route(event(3)):
    move_left
    move_left
    move_left
    move_up
    move_up
end_move_route
