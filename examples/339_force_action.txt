# force_action – Aktion für Gegner oder Held im Kampf erzwingen
# Ziel: enemy(Index) | actor(ID)
# Aktion: basic(ID) | skill(ID)
# basic: 0=angreifen, 1=nichts tun, 2=fliehen, 3=Gegenstand benutzen
# target: Ziel-Index (-1=zufällig, 0=letztes Ziel, 1+=Index)
# Timing: normal | now

# Gegner 1 greift Held 1 an (sofort)
force_action(enemy(1), basic(0), target=1, now)

# Gegner 2 führt Skill 5 aus (normal)
force_action(enemy(2), skill(5), target=-1, normal)

# Held 1 benutzt Skill 3 gegen Gegner 1
force_action(actor(1), skill(3), target=1, now)

# Held wird zur Flucht gezwungen
force_action(actor(1), basic(2), target=0, now)

# Boss-Spezialangriff bei niedrigen HP
if enemy(1) state(4):
    show_text("Der Boss rastet aus!")
    force_action(enemy(1), skill(10), target=-1, now)
endif
